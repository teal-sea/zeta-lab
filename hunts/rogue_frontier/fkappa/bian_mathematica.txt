gencomnormilist[l_, \[Kappa]_, norm_] :=
  Module[{i, j, tlen, tn, tdis, clist = {}, indexlist = {}},
   Which[l == 1 && \[Kappa] >= norm, clist = Append[clist, norm],
    l == 1 && \[Kappa] < norm, clist = {},
    l > 1,
    clist = Range[\[Kappa]];
    For[i = 1, i < l, i++,
     clist = Distribute[List[clist, Range[\[Kappa]]], List];
     clist = Flatten[clist];
     clist = Partition[clist, i + 1];
     tlen = Length[clist];
     indexlist = {};
     For[j = 1, j < tlen + 1, j++,
      tn = Total[clist[[j]]]; tdis = norm - tn;
      Which[l - i - 1 > tdis, indexlist = Append[indexlist, {j}],
       (l - i - 1) \[Kappa] < tdis, indexlist = Append[indexlist, {j}]
       ]
      ];
     clist = Delete[clist, indexlist]
     ]
    ];
   If[clist == {}, clist = False,
    If[NumberQ[clist[[1]]], clist = {clist}];
    ];
   Return[clist]
   ]

(* printed p. 102 *)

gennormzerocomlist[l_, \[Kappa]_, norm_] :=
  Module[{i, j, tlen, tn, tdis, clist = {}, indexlist = {}},
   Which[\[Kappa] == 0 && norm == 0, clist = Table[0, {l}],
    l == 1 && \[Kappa] >= norm, clist = Append[clist, norm],
    l == 1 && \[Kappa] < norm, clist = {},
    l > 1,
    clist = Range[0, \[Kappa]];
    For[i = 1, i < l, i++,
     clist = Distribute[List[clist, Range[0, \[Kappa]]], List];
     clist = Flatten[clist];
     clist = Partition[clist, i + 1];
     tlen = Length[clist];
     indexlist = {};
     For[j = 1, j < tlen + 1, j++,
      tn = Total[clist[[j]]]; tdis = norm - tn;
      Which[(l - i - 1) \[Kappa] < tdis,
       indexlist = Append[indexlist, {j}],
       tn > norm,
       indexlist = Append[indexlist, {j}]
       ]
      ];
     clist = Delete[clist, indexlist]
     ]
    ];
   If[clist == {}, clist = False,
    If[NumberQ[clist[[1]]], clist = {clist}];
    ];
   Return[clist]
   ]

mycomposition[v_] := Module[{i, j, lc, partlist = {}, complist},
   For[i = 1, i < v + 1, i++,
    complist = gencomnormilist[i, v, v];
    lc = Length[complist];
    For[j = 1, j < lc + 1, j++,
     Which[NumberQ[complist[[j]]] == True,
      complist = PadRight[complist, v];
      partlist = Append[partlist, complist],
      NumberQ[complist[[j]]] == False,
      complist[[j]] = PadRight[complist[[j]], v];
      partlist = Append[partlist, complist[[j]]]
      ]
     ]
    ]; If[partlist == {}, Print["error,mycomposition"]];
   Return[partlist]
   ]

halfdim[v_] := Floor[(v + 1)/2]

ocv[v_] := Module[{i, lv, oddcomv = {}},
   lv = Length[v];
   For[i = 1, i < lv + 1, i += 2,
    oddcomv = Append[oddcomv, v[[i]]]
    ]; If[oddcomv == {}, Print["error,ocv"]];
   Return[oddcomv]
   ]

ecv[v_] := Module[{i, lv, lhv, evencomv = {}},
   lv = Length[v]; lhv = halfdim[lv];
   For[i = 2, i < lv + 1, i += 2,
    evencomv = Append[evencomv, v[[i]]]
    ];
   evencomv = PadRight[evencomv, lhv];
   If[evencomv == {}, Print["error,ecv"]];
   Return[evencomv]
   ]

mypermutation[i_] := Permutations[Range[i]];

outervt[vtemp1_, vtemp2_] := Distribute[List[vtemp1, vtemp2], List]

inverseperm[permlist_] := Module[{ml, j, invperm},
   ml = Length[permlist]; invperm = permlist;
   For[j = 1, j < ml + 1, j++,
    invperm = ReplacePart[invperm, j, permlist[[j]]]
    ]; If[invperm == {}, Print["error,inverseperm"]];
   Return[invperm]
   ]

inversepermvec[permlist_, vec_] := Module[{invec, i, invperm, tlen},
   invperm = inverseperm[permlist];
   invec = vec;
   tlen = Length[vec];
   For[i = 1, i < tlen + 1, i++,
    invec = ReplacePart[invec, vec[[invperm[[i]]]], i]
    ]; If[invec == {}, Print["error,inversepermvec"]];
   Return[invec]
   ]

mynorm[v_] := Total[v]

mytheta[alfa1_, alfa2_] := If[mynorm[alfa1] == mynorm[alfa2], 1, 0]

mydelta[vl_, wk_] := Module[{li, lj, ni, nj, delta},
   li = Length[vl]; ni = mynorm[vl];
   lj = Length[wk]; nj = mynorm[wk];
   Which[Max[li, lj] <= Min[ni, nj], delta = 1,
    Min[ni, nj] < Max[li, lj], delta = 0
    ];
   Return[delta]
   ]

mcl[alfa_, beta_] := gennormzerocomlist[alfa, beta, beta]

mclpadd[alfa_, beta_, len_] := Module[{i, tl, templist},
   templist = mcl[alfa, beta];
   If[Length[templist] == 1,
    templist = templist[[1]];
    templist = PadRight[templist, len],
    tl = Length[templist];
    For[i = 1, i < tl + 1, i++,
     templist[[i]] = PadRight[templist[[i]], len]
     ];
    ];
   If[templist == {}, Print["error,mclpadd"]];
   Return[templist]
   ]

partvec[vec_] := Module[{i, lv, pv = {}, tlist},
   (* printed p. 107 *)
   lv = Length[vec];
   For[i = 1, i < lv + 1, i++,
    tlist = mycomposition[vec[[i]]];
    pv = Append[pv, tlist]
    ]; If[pv == {}, Print["error,partvec"]];
   Return[pv]
   ]

partvecouter[vec_] :=
  Module[{i, lv, pv = {}, tlist, j, tlen, norm, counter, templ = {},
    nl = {}, fov = {}},
   lv = Length[vec]; norm = mynorm[vec];
   tlist = mycomposition[vec[[1]]];
   pv = Append[pv, tlist];
   For[i = 2, i < lv + 1, i++,
    tlist = mycomposition[vec[[i]]];
    tlen = Length[pv];
    For[j = 1, j < tlen + 1, j++,
     pv[[j]] = Distribute[List[pv[[j]], tlist], List]
     ];
    ];
   pv = Flatten[pv];
   pv = Partition[pv, norm];
   tlen = Length[pv]; counter = 0;
   For[i = 1, i < tlen + 1, i++,
    counter = 0;
    For[j = 1, j < lv + 1, j++,
     (* printed p. 108 *)
     For[k = 1, k < vec[[j]] + 1, k++,
      templ = Append[templ, pv[[i, counter + k]]];
      ];
     counter += vec[[j]];
     nl = Append[nl, templ]; templ = {};
     ];
    fov = Append[fov, nl]; nl = {};
    ]; If[fov == {}, Print["error,partvecouter"]];
   Return[fov]
   ]

fulllist[alfa_, beta_] := Module[{i, lv, tl, flist = {}, tlist},
   lv = Length[alfa];
   tlist = mcl[alfa[[1]], beta[[1]]];
   flist = tlist;
   For[i = 2, i < lv + 1, i++,
    If[alfa[[i]] == 0, Continue[]];
    tlist = mcl[alfa[[i]], beta[[i]]];
    flist = Distribute[List[flist, tlist], List];
    ];
   tl = Length[flist];
   For[i = 1, i < tl + 1, i++,
    flist = ReplacePart[flist, Flatten[flist[[i]]], i]
    ]; If[flist == {},
    Print["error,fulllist", "alfa=:", alfa, "beta=:", beta]];
   Return[flist]
   ]

cons4vec[alfa1_, beta1_, alfa2_, beta2_] :=
  Module[{consfin, cons1, conspart1, ttao, tsigma, tnormalfa1,
    tnormalfa2, tlenalfa1, tlenalfa2, tclist, tdlist, outerlist,
    lenouterlist, i, cons2, conssum = 0, cons3, conspart3, cons4,
    cdouterlist = {}, j, cons5},
   tnormalfa1 = mynorm[alfa1]; tnormalfa2 = mynorm[alfa2];
   If[tnormalfa1 != tnormalfa2, consfin = 0,
    cons1 =
     tnormalfa1! (tnormalfa1 + mynorm[beta1] + tnormalfa2 +
         mynorm[beta2] - 1)!;
    conspart1 = 1/cons1;
    tsigma = mypermutation[tnormalfa1];
    ttao = mypermutation[tnormalfa1];
    outerlist = Distribute[List[tsigma, ttao], List];
    tclist = fulllist[alfa1, beta1];
    tdlist = fulllist[alfa2, beta2];
    cdouterlist = Distribute[List[tclist, tdlist], List];
    (* printed p. 110 *)
    lenouterlist = Length[outerlist];
    lencdouterlist = Length[cdouterlist];
    (* NOTE: lencdouterlist is not in the Module local list as printed *)
    For[i = 1, i < lenouterlist + 1, i++,
     For[j = 1, j < lencdouterlist + 1, j++,
      cons3 =
       prodfacvec[cdouterlist[[j, 1]]] prodfacvec[
         cdouterlist[[j, 2]]];
      conspart3 = 1/cons3;
      cons2 =
       prodfacvec[alfa1] prodfacvec[beta1] prodfacvec[
         alfa2] prodfacvec[beta2];
      cons4 =
       prodfacvec[
        inversepermvec[outerlist[[i, 1]], cdouterlist[[j, 1]]] +
         inversepermvec[outerlist[[i, 2]], cdouterlist[[j, 2]]] +
         idenvec[tnormalfa1]];
      conssum += cons2 conspart3 cons4;
      ];
     ];
    consfin = conspart1 conssum];
   Return[consfin]
   ]

myouter1[list1_, list2_] := Module[{egen = {}, i, len, lt},
   len = Length[list2];
   If[list1 == {},

     lt = list1;
     lt = Append[lt, list2[[i]]];
     egen = Append[egen, lt]

   Return[egen]

numberouter[list1_, list2_] :=
  Module[{len1, len2, i, j, myouter = {}, tl = {}},
   len1 = Length[list1]; len2 = Length[list2];
   If[list1 == {},
    myouter = list2,
    For[i = 1, i < len1 + 1, i++,
     For[j = 1, j < len2 + 1, j++,
      tl = {};
      tl = Append[tl, list1[[i]]];
      tl = Append[tl, list2[[j]]];
      myouter = Append[myouter, tl]
      ]
     ]
    ];
   (* printed p. 112 *)
   Return[myouter]
   ]

listouterleft[list1_, list2_] :=
  Module[{len1, len2, i, j, myouter = {}, tl = {}},
   len1 = Length[list1]; len2 = Length[list2];
   For[i = 1, i < len1 + 1, i++,
    tl = {list1[[i]]};
    tl = Append[tl, list2];
    myouter = Append[myouter, tl]
    ];
   Return[myouter]
   ]

listouterright[list1_, list2_] :=
  Module[{len1, len2, i, j, myouter = {}, tl = {}},
   len1 = Length[list1]; len2 = Length[list2];
   For[j = 1, j < len2 + 1, j++,
    tl = {list2[[j]]};
    tl = Prepend[tl, list1];
    myouter = Append[myouter, tl]
    ];
   (* printed p. 113 *)
   Return[myouter]
   ]

myouter2[list1_, list2_] :=
  Module[{len1, len2, i, j, myouter = {}, tl = {}},
   len1 = Length[list1]; len2 = Length[list2];
   Which[list1 == {},
    myouter = numberouter[list1, list2],
    Depth[list1] > Depth[list2],
    myouter = listouterleft[list1, list2],
    Depth[list1] <= Depth[list2],
    myouter = listouterright[list1, list2]
    ];
   Return[myouter]
   ]

egen[beta_] :=
  Module[{el = {}, i, j, len, lenel, fl = {}, cl = {}, lencl, tmc,
    lentmc, t},
   len = Length[beta];
   cl = Append[cl, mclpadd[1, beta[[1]], len]];
   For[i = 2, i < len + 1, i++,
    tmc = mclpadd[i, beta[[i]], len]; t = NumberQ[tmc[[1]]];
    If[t,
     cl = Distribute[List[cl, {tmc}], List],
     cl = Distribute[List[cl, tmc], List]
     ]
    ];
   cl = Flatten[cl];
   cl = Partition[cl, len];
   cl = Partition[cl, len];
   If[cl == {}, Print["error,egen"]];
   (* printed p. 115 *)
   Return[cl]
   ]

egenlast[beta_] :=
  Module[{ell = {}, i, len, f, egen = {}, lt = {}, tmc, t},
   len = Length[beta];
   ell = Append[ell, mclpadd[1, beta[[1]], len]];
   For[i = 2, i < len + 1, i++,
    tmc = mclpadd[i, beta[[i]], len]; t = NumberQ[tmc[[1]]];
    If[t,
     ell = Distribute[List[ell, {tmc}], List],
     ell = Distribute[List[ell, tmc], List]
     ]
    ];
   ell = Flatten[ell];
   ell = Partition[ell, len];
   ell = Partition[ell, len];
   f = unitvec[len];
   ell = Distribute[List[ell, f], List];
   ell = Flatten[ell];
   ell = Partition[ell, len];
   ell = Partition[ell, len + 1];

mypartition[vbl_, elgl_] :=
  Module[{lvbl, tlb, i, j, tlistb, lenel, pvbl = {}, lcounter = 0, k,
    tl1l = {}, tl2l = {}, tl3l = {}, lenpvbl, tcyles, cylen = 0, p,
    q},
   lvbl = Length[vbl]; lenel = Length[elgl];
   For[p = 1, p < lvbl, p++,
    cylen += Length[vbl[[p]]]^2
    ];
   cylen += Length[vbl[[lvbl]]]^2 + Length[vbl[[lvbl]]];
   tcyles = lenel/cylen;
   If[! IntegerQ[tcyles], Print["error", tcyles]];
   lcounter = 0;
   For[q = 1, q < tcyles + 1, q++,
    For[i = 1, i < lvbl, i++,
     tlb = Length[vbl[[i]]];
     For[j = 1, j < tlb + 1, j++,
      For[k = 1, k < tlb + 1, k++,
       tl1l = Append[tl1l, elgl[[lcounter + 1]]];
       lcounter++
       ];
      (* printed p. 117 *)
      tl2l = Append[tl2l, tl1l]; tl1l = {}
      ];
     tl3l = Append[tl3l, tl2l]; tl2l = {}
     ];
    tlb = Length[vbl[[lvbl]]]; tl1l = {}; tl2l = {};
    For[j = 1, j < tlb + 2, j++,
     For[k = 1, k < tlb + 1, k++,
      tl1l = Append[tl1l, elgl[[lcounter + 1]]]; lcounter++
      ];
     tl2l = Append[tl2l, tl1l]; tl1l = {}
     ];
    tl3l = Append[tl3l, tl2l]; tl2l = {};
    pvbl = Append[pvbl, tl3l]; tl3l = {};
    ];
   If[pvbl == {}, Print["error,mypartition"]];
   Return[pvbl]
   ]

elistgen[r_] := Module[{betalist, lenbetal, i, elg = {}, tl = {},},
   betalist = betagen[r];

    elg = Flatten[elg];

    ];

  Module[{i, lenel, felist = {}, j, tb = {}, k, tlist = {}, tlenb,

   If[Depth[elist] == 3, stanelist = Append[stanelist, elist];
    tlenvb = Length[stanelist[[1]]];
    tb = stanelist[[1, 1]];
    (* printed p. 119 *)
    For[j = 2, j < tlenvb + 1, j++,
     tb = tb + stanelist[[1, j]]
     ];
    tlist = Append[tlist, tb];
    tlist = Flatten[tlist],
    lenel = Length[elist];
    For[i = 1, i < lenel + 1, i++,
     tlenvb = Length[elist[[i]]];
     tb = elist[[i, 1]];
     For[j = 2, j < tlenvb + 1, j++,
      tb = tb + elist[[i, j]]
      ];
     tlist = Append[tlist, tb]; tb = {}
     ];
    tlist = Flatten[tlist];
    ];
   (* printed p. 120 *)
   If[tlist == {}, Print["error,betalink"]];

conse[elist_] :=
  Module[{lenel, i, j, k, l, tlen, ce = 1, fe = 1, len1, len2, len3,

sumgen[v_, w_] :=
  Module[{sumv = 0, tpv, i, lentpv, valfa, vbeta, j, tlen, vece, tnorm,
    thalflen, k, l, tlenl, compe = {}, facprodbeta = 1, facprode = 1,
    talfa = {}, tbeta = {}, sume = {}, tf, tpw, lencp, cp = {},
    sign = 1, normv, normw, lenv, lenw, trv, trw, tvalfa, tvbeta,
    twalfa, twbeta, talfalink, tbetalink, talfaplink, tbetaplink,
    (* printed p. 122 *)
    telist = {}, teplist = {}, cptep = {}, lencptep, tempelist = {},
    tempeplist = {}, tempconse, tempconsbeta, tempconsep,
    tempconsbetap, fconstemp, tc4, talfalinkfla, talfaplinfla},
   tpv = partvecouter[v]; tpw = partvecouter[w];
   cp = pvodist[tpv, tpw];
   lencp = Length[cp];
   normv = mynorm[v]; normw = mynorm[w];
   sign = (-1)^(normv + normw);
   For[i = 1, i < lencp + 1, i++,
    trv = cp[[i, 1]]; trw = cp[[i, 2]];
    talfalink = alfalink[trv]; talfalinkfla = Flatten[talfalink];
    talfaplink = alfalink[trw]; talfaplinkfla = Flatten[talfaplink];
    If[mynorm[talfalinkfla] != mynorm[talfaplinkfla], Continue[]];
    tvbeta = betagen[trv]; twbeta = betagen[trw];
    telist = elistgen[trv];
    teplist = elistgen[trw];
    (* printed p. 123 *)
    cptep = pvodist[telist, teplist];
    lencptep = Length[cptep];
    For[j = 1, j < lencptep + 1, j++,
     tempelist = cptep[[j, 1]];
     tempeplist = cptep[[j, 2]];
     tempconsbeta = consbeta[tvbeta];
     tempconsbetap = consbeta[twbeta];
     tempconse = conse[tempelist];
     tempconsep = conse[tempeplist];
     tbetalink = betalink[tempelist];
     tbetaplink = betalink[tempeplist];
     tc4 =
      cons4vec[talfalinkfla, tbetalink, talfaplinkfla, tbetaplink];
     sumv += tempconsbeta tempconsbetap tempconse tempconsep tc4;
     ];
    ];
   sumv *= sign;
   Return[sumv]
   (* printed p. 124 *)
   ]

mycompositionkapa[v_, kapa_] :=
  Module[{i, j, lc, partlist = {}, complist},
   For[i = 1, i < v + 1, i++,
    complist = gencomnormilist[i, Min[v, kapa], v];
    lc = Length[complist];
    For[j = 1, j < lc + 1, j++,
     Which[NumberQ[complist[[j]]] == True,
      complist = PadRight[complist, v];
      partlist = Append[partlist, complist],
      NumberQ[complist[[j]]] == False,
      complist[[j]] = PadRight[complist[[j]], v];
      partlist = Append[partlist, complist[[j]]]
      ]
     ]
    ]; If[partlist == {}, Print["error,mycomposition"]];
   Return[partlist]
   ]

cons4veckapa[alfa1_, beta1_, alfa2_, beta2_, kapa_] :=
  Module[{consfin, cons1, conspart1, ttao, tsigma, tnormalfa1,
    tnormalfa2, tlenalfa1, tlenalfa2, tclist, tdlist, outerlist,
    lenouterlist, i, cons2, conssum = 0, cons3, conspart3, cons4,
    cdouterlist = {}, j, cons5},
   tnormalfa1 = mynorm[alfa1]; tnormalfa2 = mynorm[alfa2];
   If[tnormalfa1 != tnormalfa2, consfin = 0,
    (* printed p. 136 *)
    cons1 =
     tnormalfa1! (tnormalfa1 + mynorm[beta1] + tnormalfa2 +
         mynorm[beta2] - 1)!;
    conspart1 = 1/cons1;
    tsigma = mypermutation[tnormalfa1];
    ttao = mypermutation[tnormalfa1];
    outerlist = Distribute[List[tsigma, ttao], List];
    tclist = fulllistkapa[alfa1, beta1, kapa];
    tdlist = fulllistkapa[alfa2, beta2, kapa];
    cdouterlist = Distribute[List[tclist, tdlist], List];
    lenouterlist = Length[outerlist];
    lencdouterlist = Length[cdouterlist];
    For[i = 1, i < lenouterlist + 1, i++,
     For[j = 1, j < lencdouterlist + 1, j++,
      cons3 =
       prodfacvec[cdouterlist[[j, 1]]] prodfacvec[
         cdouterlist[[j, 2]]];
      conspart3 = 1/cons3;
      cons2 =
       prodfacvec[alfa1] prodfacvec[beta1] prodfacvec[
         alfa2] prodfacvec[beta2];
      cons4 =
       prodfacvec[
        inversepermvec[outerlist[[i, 1]], cdouterlist[[j, 1]]] +
         (* printed p. 137 *)
         inversepermvec[outerlist[[i, 2]], cdouterlist[[j, 2]]] +
         idenvec[tnormalfa1]];
      conssum += cons2 conspart3 cons4;
      ];
     ];
    consfin = conspart1 conssum
    ];
   Return[consfin]
   ]
